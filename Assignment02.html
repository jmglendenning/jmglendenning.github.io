  1. Collect the same data with gtrendsR
library(gtrendsR)
library(readr)

TrumpHarrisElection <- gtrends(
  c("Trump", "Harris", "election"),
  onlyInterest = TRUE,
  geo = "US",
  gprop = "web",
  time = "today+5-y",
  category = 0
)

the_df <- TrumpHarrisElection$interest_over_time
plot(TrumpHarrisElection)
# Cache the raw pull because later calls may return slightly different values.
write_csv(
  the_df,
  paste0("gtrends_us_5y_", Sys.Date(), ".csv")
)
