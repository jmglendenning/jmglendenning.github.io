# -------------------------------------------------------------------
# Base API request

govinfo <- function(path) {
  request(paste0("https://api.govinfo.gov/", path)) |>
    req_url_query(api_key = key) |>
    req_retry(max_tries = 3)
}

# -------------------------------------------------------------------
# Search Senate Foreign Relations Committee hearings
# -------------------------------------------------------------------

search_page <- function(offset = "*") {

  govinfo("search") |>
    req_body_json(list(
      query = paste(
        "collection:CHRG",
        "AND chamber:senate",
        'AND committee:("foreign relations")'
      ),
      pageSize = page_size,
      offsetMark = offset,
      sorts = list(
        list(
          field = "publishdate",
          sortOrder = "DESC"
        )
      )
    )) |>
    req_perform() |>
    resp_body_json()
}

# -------------------------------------------------------------------
# Retrieve package metadata
# -------------------------------------------------------------------

get_summary <- function(id) {

  tryCatch({

    govinfo(paste0("packages/", id, "/summary")) |>
      req_perform() |>
      resp_body_json()

  }, error = function(e) {

    warning(
      "Could not retrieve package ", id,
      ": ", conditionMessage(e)
    )

    NULL
  })
}

# -------------------------------------------------------------------
# Search for ten qualifying hearings
# -------------------------------------------------------------------

selected_hearings <- list()
seen <- character()
offset <- "*"

while (
  length(selected_hearings) < n_target &&
  length(seen) < max_candidates
) {

  page <- search_page(offset)

  results <- page$results

  if (is.null(results) || length(results) == 0) {
    break
  }

  # Process search results
  for (result in results) {

    if (
      length(selected_hearings) >= n_target ||
      length(seen) >= max_candidates
    ) {
      break
    }

    id <- result$packageId

    if (
      is.null(id) ||
      !nzchar(id) ||
      id %in% seen
    ) {
      next
    }

    seen <- c(seen, id)

    # Senate hearing package IDs contain "shrg"
    if (!grepl("^CHRG-[0-9]+shrg", id, ignore.case = TRUE)) {
      next
    }

    # Retrieve metadata
    summary <- get_summary(id)

    Sys.sleep(0.5)

    if (is.null(summary)) {
      next
    }

    # Retain qualifying hearing
    selected_hearings[[id]] <- summary

    # Progress reporting
    cat(
      "Examined:", length(seen),
      "| Found:", length(selected_hearings),
      "of", n_target, "\n"
    )

    if (length(selected_hearings) >= n_target) {
      break
    }
  }

  # Stop if complete or capped
  if (
    length(selected_hearings) >= n_target ||
    length(seen) >= max_candidates
  ) {
    break
  }

  # Move to next page
  next_offset <- page$offsetMark

  if (
    is.null(next_offset) ||
    identical(next_offset, offset)
  ) {
    break
  }

  offset <- next_offset
}

# Download function
download_pdf <- function(id, url) {

  dest <- file.path(pdf_dir, paste0(id, ".pdf"))

  started <- format(
    Sys.time(),
    "%Y-%m-%d %H:%M:%S %Z"
  )

  if (is.na(url) || !nzchar(url)) {

    result <- list(
      status = "no pdfLink",
      http = NA_integer_,
      error = NA_character_
    )

  } else {

    result <- tryCatch({

      response <- request(url) |>
        req_url_query(api_key = key) |>
        req_retry(max_tries = 3) |>
        req_perform(path = dest)

      list(
        status = "ok",
        http = resp_status(response),
        error = NA_character_
      )

    }, error = function(e) {

      list(
        status = "failed",
        http = NA_integer_,
        error = conditionMessage(e)
      )
    })
  }

  # Pause between downloads
  Sys.sleep(runif(1, 1, 3))

  tibble(
    packageId = id,
    pdf_url = url,
    file = dest,
    status = result$status,
    http_status = result$http,
    bytes = if (result$status == "ok") {
      file.size(dest)
    } else {
      NA_real_
    },
    error = result$error,
    started = started
  )
}

4. Report

Write a short report on the difficulties you met.

  1. How usable are the downloaded documents as data? What would it take to turn ten PDFs into a table or a corpus?

The PDFs are useful as sources but are not immediately useable for quantitative analysis. Converting them into a corpus would require extracting the text, removing formatting artifacts, and standardizing the documents. Scanned PDFs may also require OCR. Creating a structured table would require identifying fields such as hearing dates, committee names, speakers, and testimony, then validating the extracted information. This may currently be possible with some AI models.

  1. How would you improve the process? Consider the other formats in download, and what the metadata gives you without opening a file.

The process could be improved by downloading text or XML files rather than PDFs if available. These formats are easier to parse and would reduce the need for text extraction or OCR. The API metadata already provides information such as titles, dates, IDs, and links, which allows the documents to be indexed and filtered. The main difficulties encountered were handling nested JSON responses, identifying the correct committee hearings, and managing pagination.

The largest problem was that the package summaries for these hearings listed only premisLink, zipLink and modsLink, with no pdfLink, so the first run downloaded nothing. Requesting each package’s PDF directly from /packages/{packageId}/pdf fixed it.

  1. In two sentences: what did the API give you that the search page and last week’s scraping did not?

The API provided structured, machine-readable results. The API allowed direct access to document metadata and download links.

    destfile <- paste0(save_dir, "govfiles_", id, ".pdf")
    download.file(url, destfile = destfile, mode = "wb") # Binary files
    Sys.sleep(runif(1, 1, 3))  # Important: random sleep between 1 and 3 seconds to avoid suspicion of "hacking" the server
    return(paste("Successfully downloaded:", url))
  },
  error = function(e) {
    return(paste("Failed to download:", url))
  })
}

# Base request: every call goes to api.govinfo.gov and carries api_key
govinfo <- function(path) {
  request("https://api.govinfo.gov") |>
    req_url_path_append(path) |>
    req_url_query(api_key = api_key) |>
    req_retry(max_tries = 3)
}

B2. Search, page, and keep the first ten Senate Foreign Relations hearings

The full-text query also matches hearings of other committees that merely mention the Committee on Foreign Relations, so each candidate’s summary is checked for the committee itself.

search_page <- function(offset) {
  govinfo("search") |>
    req_body_json(list(
      query      = 'collection:(CHRG) AND "Committee on Foreign Relations"',
      pageSize   = 100,
      offsetMark = offset,
      sorts      = list(list(field = "publishdate", sortOrder = "DESC"))
    )) |>
    req_perform() |>
    resp_body_json()
}

get_summary <- function(id) {
  govinfo(paste0("packages/", id, "/summary")) |>
    req_perform() |>
    resp_body_json()
}

is_sfrc <- function(s) {
  any(map_lgl(s$committees %||% list(), \(cm)
    identical(toupper(cm$chamber %||% ""), "SENATE") &&
      grepl("Foreign Relations", cm$committeeName %||% "", ignore.case = TRUE)))
}

n_target <- 10
hearings <- list()
seen     <- character()
offset   <- "*"

while (length(hearings) < n_target) {
  page <- search_page(offset)
  if (length(page$results) == 0) break          # no more results: stop

  for (r in page$results) {
    id <- r$packageId
    if (is.null(id) || id %in% seen) next        # skip granules already seen
    seen <- c(seen, id)
    s <- get_summary(id)
    Sys.sleep(0.5)
    if (is_sfrc(s)) hearings[[id]] <- s
    if (length(hearings) == n_target) break
  }

  if (is.null(page$offsetMark) || identical(page$offsetMark, offset)) break
  offset <- page$offsetMark                      # send it back for the next page
}

message(length(hearings), " hearings found after checking ", length(seen), " packages")

B4. Download the PDFs and write the log

download_pdf <- function(id, url) {
  dest    <- file.path(pdf_dir, paste0(id, ".pdf"))
  started <- format(Sys.time(), "%Y-%m-%d %H:%M:%S %Z")

  res <- if (is.na(url)) {
    list(status = "no pdfLink", http = NA_integer_, error = NA_character_)
  } else {
    tryCatch({
      resp <- request(url) |>
        req_url_query(api_key = api_key) |>
        req_retry(max_tries = 3) |>
        req_perform(path = dest)
      list(status = "ok", http = resp_status(resp), error = NA_character_)
    }, error = \(e) list(status = "failed", http = NA_integer_,
                         error = conditionMessage(e)))
  }
  Sys.sleep(runif(1, 1, 3))  # polite pause, as in govtdata01.R

  tibble(
    packageId   = id,
    pdf_url     = url,                       # without the key
    file        = dest,
    status      = res$status,
    http_status = res$http,
    bytes       = if (res$status == "ok") file.size(dest) else NA_real_,
    error       = res$error,
    started     = started
  )
}
